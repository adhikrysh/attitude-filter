cmake_minimum_required(VERSION 3.20)
project(attitude_filter VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
find_package(Eigen3 REQUIRED NO_MODULE)
option(ATTITUDE_SANITIZERS "Enable address and undefined-behavior checks" OFF)
add_library(attitude src/filter.cpp src/rotation.cpp)
target_include_directories(attitude PUBLIC include)
target_link_libraries(attitude PUBLIC Eigen3::Eigen)
target_compile_features(attitude PUBLIC cxx_std_20)
if(ATTITUDE_SANITIZERS)
  target_compile_options(attitude PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(attitude PUBLIC -fsanitize=address,undefined)
endif()
add_executable(attitude-demo src/main.cpp)
target_link_libraries(attitude-demo PRIVATE attitude)
enable_testing()
add_executable(attitude-tests tests/test_filter.cpp)
target_link_libraries(attitude-tests PRIVATE attitude)
add_test(NAME estimation-and-jacobians COMMAND attitude-tests)
